// プルダウンに出したい候補一覧
const candidates = [
  "1.1 SELECT",
  "1.2 INSERT・UPDATE・DELETE",
  "1.3 CREATE",
  "1.4 権限",
  "1.5 プログラム言語におけるSQL文",
  "1.6 SQL文暗記チェックシート",
  "2.1 情報処理試験の中の概念データモデル",
  "2.2 E-R図（拡張E-R図）",
  "2.3 様々なビジネスモデル",
  "3.1 関係スキーマの表記方法",
  "3.2 関数従属性",
  "3.3 キー",
  "3.4 正規化",
  "4.1 表領域",
  "4.2 トランザクション管理機能",
  "4.3 障害回復機能",
  "4.4 索引（インデックス）",
  "4.5 分散データベース",
  "4.6 テーブルの物理分割",
  "4.7 クラスタ構成",
  "演習"
];

// メモをノートタイトル（[[...]]）として使う際、ファイル名で使えない文字や余分な空白を除去する関数
const sanitizeLinkTitle = (rawText) =>
  rawText
    .replace(/[\\/:*?"<>|]/g, "") // Obsidian/OSで禁止されている記号を削除
    .replace(/\s+/g, " ")          // 連続する空白・改行を半角スペース1つにまとめる
    .trim();                       // 前後の空白を削除

// 現在開いているノートファイルを取得
const tfile = app.workspace.getActiveFile();

// プルダウン・メモ欄・追加ボタンをまとめるコンテナ要素を生成
const container = dv.el("div", "", { cls: "todo-dropdown-box" });

// プルダウン（<select>）を作成し、初期オプションと候補を追加
const selectEl = container.createEl("select");
selectEl.createEl("option", { text: "-- やることを選択 --", value: "" });
for (const item of candidates) {
  selectEl.createEl("option", { text: item, value: item });
}

// 端的にメモを入力するためのテキストボックスを作成
const memoEl = container.createEl("input", {
  type: "text",
  placeholder: "メモ（任意・[[リンク]]化）"
});
memoEl.style.marginLeft = "8px";
memoEl.style.width = "220px";

// 追加ボタンを作成
const addBtn = container.createEl("button", { text: "追加" });
addBtn.style.marginLeft = "8px";

// 選択項目＋メモ（あれば[[リンク]]）を、このコードブロック（追加UI）の直下にチェックボックスとして挿入する処理
const appendTask = async () => {
  const chosen = selectEl.value;
  // 項目未選択、または対象ノートが取得できない場合は何もしない
  if (!chosen || !tfile) return;

  // メモを整形し、空でなければ [[メモ]] 形式のリンクタイトルに変換する
  const cleanMemo = sanitizeLinkTitle(memoEl.value);
  const taskLabel = cleanMemo ? `${chosen} [[${cleanMemo}]]` : chosen;
  const newTaskLine = `- [ ] ${taskLabel}`;

  // 現在のノート本文を行単位で読み込む
  const content = await app.vault.read(tfile);
  const lines = content.split(/\r?\n/);

  // 現在の ```dataviewjs コードブロックの終了行（```）を探し、その次の行を挿入位置にする
  // これにより、チェックリストが常に「追加」UIの下に並ぶようになる
  let insertIndex = lines.length;
  for (let i = 0; i < lines.length; i++) {
    if (lines[i].trim().startsWith("```dataviewjs")) {
      for (let j = i + 1; j < lines.length; j++) {
        if (lines[j].trim() === "```") {
          insertIndex = j + 1; // 終了 ``` の直後
          break;
        }
      }
      break;
    }
  }

  // チェックリストの上に1行空行をあけて挿入（すでに直上が空行なら重複させない）
  const linesToInsert = lines[insertIndex - 1] === "" ? [newTaskLine] : ["", newTaskLine];
  lines.splice(insertIndex, 0, ...linesToInsert);
  await app.vault.modify(tfile, lines.join("\n"));

  // 次の入力に備えてプルダウンとメモ欄をリセットし、メモ欄にフォーカスを戻す
  selectEl.value = "";
  memoEl.value = "";
  memoEl.focus();
};

// 「追加」ボタンクリック時にタスクを追加
addBtn.addEventListener("click", appendTask);

// メモ入力欄で Enter キーを押したときもタスクを追加（フォーム送信などの既定動作は抑止）
memoEl.addEventListener("keydown", (e) => {
  if (e.key === "Enter") {
    e.preventDefault();
    appendTask();
  }
});
