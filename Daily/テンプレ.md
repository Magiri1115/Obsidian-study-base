# 簡易ダッシュボード
```dataviewjs
const page = dv.current();
// デイリーノート内のリンクをすべて取得
const links = (page.file.outlinks ?? []).map(l => l.path);
const uniquePaths = [...new Set(links)];

const rows = [];
for (const path of uniquePaths) {
  const target = dv.page(path);
  if (!target) continue;

  // 演習・解き直しの進捗（未復習 / 復習中 / 完了）
  const statusVal = String(target.status ?? "").trim() || "-";

  // #QA（大文字小文字問わず、および #QA/... / .../QA）でQAの有無を判別
  const tags = target.file.tags ?? [];
  const hasQa = tags.some(t => {
    const normalized = String(t).replace(/^#/, "").toLowerCase();
    return (
      normalized === "qa" ||
      normalized.startsWith("qa/") ||
      normalized.endsWith("/qa") ||
      normalized.includes("/qa/")
    );
  });

  rows.push([
    target.file.link,
    statusVal,
    hasQa ? "あり" : "なし"
  ]);
}

dv.table(["問題リンク", "status", "QAの有無"], rows);
```
