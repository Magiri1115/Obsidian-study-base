
# やること


```dataviewjs
// 追記先の見出し名
const headingName = "やること";

// 挿入するテンプレートノート名（拡張子なしでもOK）
const templateNoteName = "調査ノート_テンプレ";

// 新規ノートを作成するフォルダ（例: "ANA_sMILE/勉強" や "" ならルート/現在ノートと同じ場所）
const targetFolder = "";

// 種別リスト
const options = ["共有", "進捗", "QA確認", "タスク","レビュ", "その他"];

const tfile = app.workspace.getActiveFile();

// 右端のコードブロック操作ボタン（</>）と被りにくくするため右余白と上限幅を確保
const box = dv.el("div", "", { cls: "quick-memo-bar" });
box.style.display = "flex";
box.style.flexWrap = "nowrap";
box.style.gap = "8px";
box.style.alignItems = "center";
box.style.width = "100%";
box.style.maxWidth = "calc(100% - 4.8rem)";
box.style.boxSizing = "border-box";
box.style.margin = "8px 0 10px 0";

const sel = box.createEl("select");
sel.style.flexShrink = "0";
sel.createEl("option", { text: "-- 種別を選択 --", value: "" });
for (const opt of options) {
  sel.createEl("option", { text: opt, value: opt });
}

const input = box.createEl("input", {
  type: "text",
  placeholder: "簡潔なメモ（例：移行ツール 〇〇について → そのまま[[リンク]]＆テンプレノート作成）"
});
input.style.flex = "1";
input.style.minWidth = "0";
input.style.maxWidth = "72rem";
input.style.width = "100%";
input.style.boxSizing = "border-box";

const addBtn = box.createEl("button", { text: "追加" });
addBtn.style.flexShrink = "0";

// すでに [[...]] 形式ならそのまま、そうでなければ [[...]] 化する
const toWikiLink = (raw) => {
  const trimmed = raw.trim();
  if (!trimmed) return "";
  if (/^\[\[.*\]\]$/.test(trimmed)) return trimmed;
  return `[[${trimmed}]]`;
};

// [[ノート名]] からファイル名部分を抽出（エイリアス [[A|B]] にも対応）
const extractNoteTitle = (raw) => {
  const inner = raw.trim().replace(/^\[\[\vert{}\]\]$/g, "");
  const [target] = inner.split("|");
  return target.trim();
};

// テンプレートノートの内容を取得
const getTemplateContent = async (name) => {
  const candidates = [
    `${name}.md`,
    name
  ];
  for (const path of candidates) {
    const f = app.vault.getAbstractFileByPath(path);
    if (f && f.stat) {
      return await app.vault.read(f);
    }
  }
  const allMd = app.vault.getMarkdownFiles();
  const matched = allMd.find(f => f.basename === name);
  if (matched) {
    return await app.vault.read(matched);
  }
  return "";
};

// テンプレ内容をもとに新規ノートを作成（既存があればそのまま）
const ensureNoteFromTemplate = async (noteTitle) => {
  if (!noteTitle) return;

  // ファイル名に使えない記号を軽くサニタイズ
  const safeTitle = noteTitle.replace(/[\\/:*?"<>|]/g, "_");
  const folderPrefix = targetFolder
    ? targetFolder.replace(/^\/+|\/+$/g, "") + "/"
    : (tfile?.parent?.path && tfile.parent.path !== "/" ? tfile.parent.path + "/" : "");

  const fullPath = `${folderPrefix}${safeTitle}.md`;
  const existing = app.vault.getAbstractFileByPath(fullPath);
  if (existing) return existing;

  if (folderPrefix && !app.vault.getAbstractFileByPath(folderPrefix.slice(0, -1))) {
    await app.vault.createFolder(folderPrefix.slice(0, -1)).catch(() => {});
  }

  const tplContent = await getTemplateContent(templateNoteName);
  return await app.vault.create(fullPath, tplContent);
};

const appendMemo = async () => {
  if (!tfile) return;

  const rawText = input.value.trim();
  if (!rawText && !sel.value) return;

  const tag = sel.value ? `**${sel.value}** ` : "";
  const linkPart = rawText ? toWikiLink(rawText) : "";
  const noteTitle = rawText ? extractNoteTitle(rawText) : "";

  if (noteTitle) {
    await ensureNoteFromTemplate(noteTitle);
  }

  // チェックボックス形式で追記行を作成
  const body = `${tag}${linkPart}`.trim();
  const line = `- [ ] ${body}`;

  const content = await app.vault.read(tfile);
  const lines = content.split("\n");

  const headingRegex = new RegExp(`^#{1,6}\\s+${headingName}\\s*$`);
  const hIdx = lines.findIndex(l => headingRegex.test(l.trim()));

  let insertIdx = lines.length;
  if (hIdx !== -1) {
    insertIdx = hIdx + 1;
    while (insertIdx < lines.length) {
      const cur = lines[insertIdx].trim();
      if (/^#{1,6}\s+/.test(cur)) break;
      insertIdx++;
    }
    while (insertIdx > hIdx + 1 && lines[insertIdx - 1].trim() === "") {
      insertIdx--;
    }
  }

  // 1行空けて書くために、直前が空行でなければ空行を1つ挟む
  const toInsert = [];
  if (insertIdx > 0 && lines[insertIdx - 1].trim() !== "") {
    toInsert.push("");
  }
  toInsert.push(line);

  lines.splice(insertIdx, 0, ...toInsert);
  await app.vault.modify(tfile, lines.join("\n"));

  sel.value = "";
  input.value = "";
};

addBtn.addEventListener("click", appendMemo);
input.addEventListener("keydown", (e) => {
  if (e.key === "Enter") {
    e.preventDefault();
    appendMemo();
  }
});
```
