// 各プロパティの初期選択肢
const categoryOptions = ["午前1", "午前2", "午後1", "午後2", "過去問演習", "振り返り"];
const topicOptions = ["論理設計", "物理設計", "SQL", "トランザクション", "障害回復"];
const statusOptions = ["未復習", "復習中", "完了", "QA"];

const tfile = app.workspace.getActiveFile();
const cache = tfile ? app.metadataCache.getFileCache(tfile) : null;
const fm = cache?.frontmatter ?? {};

// 縦並びのコンテナを作成
const box = dv.el("div", "", { cls: "prop-dropdown-vertical" });
box.style.display = "flex";
box.style.flexDirection = "column";
box.style.gap = "8px";
box.style.margin = "8px 0";

// 各行を「ラベル（固定幅）＋ コントロール」で並べるヘルパー
const createRow = (labelText) => {
  const row = box.createEl("div");
  row.style.display = "flex";
  row.style.alignItems = "center";
  row.style.gap = "8px";

  const label = row.createEl("span", { text: labelText });
  label.style.width = "80px";
  label.style.fontWeight = "500";

  return row;
};

// プルダウン生成ヘルパー
const buildSelect = (parentRow, options, currentVal) => {
  const sel = parentRow.createEl("select");
  sel.style.minWidth = "150px";
  sel.createEl("option", { text: "-- 選択 --", value: "" });

  const merged = currentVal && !options.includes(currentVal) ? [currentVal, ...options] : [...options];
  for (const opt of merged) {
    const optionEl = sel.createEl("option", { text: opt, value: opt });
    if (opt === currentVal) optionEl.selected = true;
  }
  return sel;
};

// 1行目: category
const catRow = createRow("category");
const catSel = buildSelect(catRow, categoryOptions, fm.category ?? "");

// 2行目: topic（プルダウン + 新規追加入力欄 + 追加ボタン）
const topRow = createRow("topic");
const topSel = buildSelect(topRow, topicOptions, fm.topic ?? "");

const newTopicInput = topRow.createEl("input", {
  type: "text",
  placeholder: "新規topicを追加"
});
newTopicInput.style.width = "150px";

const addTopicBtn = topRow.createEl("button", { text: "+追加" });

const addNewTopic = () => {
  const val = newTopicInput.value.trim();
  if (!val) return;

  const exists = [...topSel.options].some(o => o.value === val);
  if (!exists) {
    topSel.createEl("option", { text: val, value: val });
  }
  topSel.value = val;
  newTopicInput.value = "";
};

addTopicBtn.addEventListener("click", addNewTopic);
newTopicInput.addEventListener("keydown", (e) => {
  if (e.key === "Enter") {
    e.preventDefault();
    addNewTopic();
  }
});

// 3行目: status
const statRow = createRow("status");
const statSel = buildSelect(statRow, statusOptions, fm.status ?? "");

// 4行目: プロパティ反映ボタン
const actionRow = createRow("");
const applyBtn = actionRow.createEl("button", { text: "プロパティ反映" });

// フロントマターと #QA タグを更新する処理
const updateFrontmatter = async () => {
  if (!tfile) return;

  await app.fileManager.processFrontMatter(tfile, (frontmatter) => {
    if (catSel.value) frontmatter.category = catSel.value;
    if (topSel.value) frontmatter.topic = topSel.value;
    if (statSel.value) frontmatter.status = statSel.value;

    const effectiveStatus = statSel.value || frontmatter.status || "";
    const isQa = effectiveStatus === "QA";

    let currentTags = [];
    if (Array.isArray(frontmatter.tags)) {
      currentTags = [...frontmatter.tags];
    } else if (typeof frontmatter.tags === "string" && frontmatter.tags.trim()) {
      currentTags = frontmatter.tags.split(/[,\s]+/).filter(Boolean);
    }

    const withoutQa = currentTags.filter(
      t => String(t).replace(/^#/, "").toLowerCase() !== "qa"
    );

    const nextTags = isQa ? [...withoutQa, "QA"] : withoutQa;

    if (nextTags.length > 0) {
      frontmatter.tags = nextTags;
    } else {
      delete frontmatter.tags;
    }
  });
};

applyBtn.addEventListener("click", updateFrontmatter);

- **`topic`**：`論理設計` / `物理設計` / `SQL` / `トランザクション` / `障害回復` など（単元・テーマ）

- **`status`**：`未復習` / `復習中` / `完了` （演習・理解の進捗）
