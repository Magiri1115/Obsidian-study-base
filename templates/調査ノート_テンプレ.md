const tfile = app.workspace.getActiveFile();
if (!tfile) {
  dv.paragraph("対象ノートが開かれていません。");
  return;
}

const cache = app.metadataCache.getFileCache(tfile);
const fm = cache?.frontmatter ?? {};

const categoryOptions = [
  "クレジットカード情報の参照",
  "クレジットカード番号移行ツール",
  "提携会員番号移行ツール",
  "デトークン化",
  "トークン化",
  "DMラベル印刷"
];
const positionOptions = ["要件定義", "設計", "実装", "テスト", "結合テスト"];
const phaseOptions = ["UT", "FT1", "FT1.5", "FT2"];
const topicOptions = ["論理設計", "物理設計", "SQL", "トランザクション", "障害回復", "sh起動"];
const statusOptions = ["未対応", "対応中", "完了", "QA"];

const getInitVal = (key) => {
  const val = fm[key];
  if (typeof val === "string" && val.trim()) {
    return val.trim();
  }
  return "";
};

const box = dv.el("div", "", { cls: "prop-setter-box" });
box.style.display = "flex";
box.style.flexDirection = "column";
box.style.gap = "8px";
box.style.margin = "8px 0 12px 0";

const selects = {};

const createPropRow = (propKey, options) => {
  const row = box.createEl("div");
  row.style.display = "flex";
  row.style.alignItems = "center";
  row.style.gap = "10px";

  const label = row.createEl("span", { text: propKey });
  label.style.minWidth = "80px";
  label.style.fontWeight = "bold";

  const sel = row.createEl("select");
  sel.style.minWidth = "220px";

  sel.createEl("option", { text: "-- 未選択 --", value: "" });

  const currentVal = getInitVal(propKey);
  const mergedOptions = currentVal && !options.includes(currentVal)
    ? [currentVal, ...options]
    : options;

  for (const opt of mergedOptions) {
    const el = sel.createEl("option", { text: opt, value: opt });
    if (opt === currentVal) el.selected = true;
  }

  selects[propKey] = sel;
};

createPropRow("category", categoryOptions);
createPropRow("position", positionOptions);
createPropRow("phase", phaseOptions);
createPropRow("topic", topicOptions);
createPropRow("status", statusOptions);

const actionRow = box.createEl("div");
actionRow.style.marginTop = "4px";

const updateBtn = actionRow.createEl("button", { text: "一括更新" });

updateBtn.addEventListener("click", async () => {
  const nextValues = {
    category: selects.category.value,
    position: selects.position.value,
    phase: selects.phase.value,
    topic: selects.topic.value,
    status: selects.status.value
  };

  await app.fileManager.processFrontMatter(tfile, (frontmatter) => {
    for (const [key, val] of Object.entries(nextValues)) {
      if (val === "") {
        delete frontmatter[key];
      } else {
        frontmatter[key] = val;
      }
    }

    // status が QA のとき tags に QA を追加し、それ以外なら QA を除去
    const existingTags = Array.isArray(frontmatter.tags)
      ? frontmatter.tags.map(String)
      : typeof frontmatter.tags === "string" && frontmatter.tags.trim()
        ? [frontmatter.tags.trim()]
        : [];

    const filteredTags = existingTags.filter(t => t !== "QA");
    if (nextValues.status === "QA") {
      filteredTags.push("QA");
    }

    if (filteredTags.length > 0) {
      frontmatter.tags = filteredTags;
    } else if (Array.isArray(frontmatter.tags)) {
      frontmatter.tags = [];
    } else {
      delete frontmatter.tags;
    }
  });
});
