const tfile = app.workspace.getActiveFile();
if (!tfile) {
  dv.paragraph("対象ノートが開かれていません。");
  return;
}

const cache = app.metadataCache.getFileCache(tfile);
const fm = cache?.frontmatter ?? {};

const categoryOptions = [
  "クレジットカード情報の参照",
  "クレジットカード番号移行ツール",
  "提携会員番号移行ツール",
  "デトークン化",
  "トークン化",
  "DMラベル印刷"
];
const positionOptions = ["要件定義", "設計", "実装", "テスト", "結合テスト"];
const phaseOptions = ["UT", "FT1", "FT1.5", "FT2"];
const statusOptions = ["未対応", "対応中", "完了", "QA"];

// topicの初期値（localStorageに保存されていればそちらを優先）
const STORAGE_KEY = "dv-custom-topic-options";
const defaultTopicOptions = [];
let topicOptions = defaultTopicOptions;
try {
  const saved = localStorage.getItem(STORAGE_KEY);
  if (saved) topicOptions = JSON.parse(saved);
} catch (e) {
  console.error(e);
}

const saveTopicOptions = (opts) => {
  topicOptions = opts;
  localStorage.setItem(STORAGE_KEY, JSON.stringify(opts));
};

const getInitVal = (key) => {
  const val = fm[key];
  if (typeof val === "string" && val.trim()) {
    return val.trim();
  }
  return "";
};

const box = dv.el("div", "", { cls: "prop-setter-box" });
box.style.display = "flex";
box.style.flexDirection = "column";
box.style.gap = "8px";
box.style.margin = "8px 0 12px 0";

const selects = {};

// プルダウンの中身を再描画する関数
const renderSelectOptions = (sel, options, selectedVal) => {
  sel.innerHTML = "";
  sel.createEl("option", { text: "-- 未選択 --", value: "" });

  const mergedOptions = selectedVal && !options.includes(selectedVal)
    ? [selectedVal, ...options]
    : options;

  for (const opt of mergedOptions) {
    const el = sel.createEl("option", { text: opt, value: opt });
    if (opt === selectedVal) el.selected = true;
  }
};

const createPropRow = (propKey, options) => {
  const row = box.createEl("div");
  row.style.display = "flex";
  row.style.alignItems = "center";
  row.style.gap = "10px";

  const label = row.createEl("span", { text: propKey });
  label.style.minWidth = "80px";
  label.style.fontWeight = "bold";

  const sel = row.createEl("select");
  sel.style.minWidth = "220px";

  renderSelectOptions(sel, options, getInitVal(propKey));
  selects[propKey] = sel;

  // topicの場合のみ追加・削除ボタンを作成
  if (propKey === "topic") {
    const addBtn = row.createEl("button", { text: "追加" });
    const delBtn = row.createEl("button", { text: "削除" });

    // 入力欄エリア（追加ボタン押下時に表示）
    const inputWrapper = row.createEl("div");
    inputWrapper.style.display = "none";
    inputWrapper.style.alignItems = "center";
    inputWrapper.style.gap = "4px";

    const input = inputWrapper.createEl("input", {
      type: "text",
      placeholder: "新しいTopic名"
    });
    input.style.width = "140px";

    const confirmBtn = inputWrapper.createEl("button", { text: "確定" });
    const cancelBtn = inputWrapper.createEl("button", { text: "×" });

    // 「追加」ボタン押下
    addBtn.addEventListener("click", () => {
      inputWrapper.style.display = "flex";
      addBtn.style.display = "none";
      delBtn.style.display = "none";
      input.value = "";
      input.focus();
    });

    // 追加のキャンセル処理
    const closeInput = () => {
      inputWrapper.style.display = "none";
      addBtn.style.display = "inline-block";
      delBtn.style.display = "inline-block";
    };
    cancelBtn.addEventListener("click", closeInput);

    // 「確定」ボタンまたはEnterキーで追加実行
    const commitAdd = () => {
      const newVal = input.value.trim();
      if (!newVal) return;

      if (!topicOptions.includes(newVal)) {
        saveTopicOptions([...topicOptions, newVal]);
      }
      renderSelectOptions(sel, topicOptions, newVal);
      closeInput();
    };

    confirmBtn.addEventListener("click", commitAdd);
    input.addEventListener("keydown", (e) => {
      if (e.key === "Enter") commitAdd();
      if (e.key === "Escape") closeInput();
    });

    // 「削除」ボタン押下（現在選択中の項目をリストから削除）
    delBtn.addEventListener("click", () => {
      const target = sel.value;
      if (!target) return;

      const nextOpts = topicOptions.filter((opt) => opt !== target);
      saveTopicOptions(nextOpts);
      renderSelectOptions(sel, topicOptions, "");
    });
  }
};

createPropRow("category", categoryOptions);
createPropRow("position", positionOptions);
createPropRow("phase", phaseOptions);
createPropRow("topic", topicOptions);
createPropRow("status", statusOptions);

const actionRow = box.createEl("div");
actionRow.style.marginTop = "4px";

const updateBtn = actionRow.createEl("button", { text: "一括更新" });

updateBtn.addEventListener("click", async () => {
  const nextValues = {
    category: selects.category.value,
    position: selects.position.value,
    phase: selects.phase.value,
    topic: selects.topic.value,
    status: selects.status.value
  };

  await app.fileManager.processFrontMatter(tfile, (frontmatter) => {
    for (const [key, val] of Object.entries(nextValues)) {
      if (val === "") {
        delete frontmatter[key];
      } else {
        frontmatter[key] = val;
      }
    }

    // status が QA のとき tags に QA を追加し、それ以外なら QA を除去
    const existingTags = Array.isArray(frontmatter.tags)
      ? frontmatter.tags.map(String)
      : typeof frontmatter.tags === "string" && frontmatter.tags.trim()
        ? [frontmatter.tags.trim()]
        : [];

    const filteredTags = existingTags.filter(t => t !== "QA");
    if (nextValues.status === "QA") {
      filteredTags.push("QA");
    }

    if (filteredTags.length > 0) {
      frontmatter.tags = filteredTags;
    } else if (Array.isArray(frontmatter.tags)) {
      frontmatter.tags = [];
    } else {
      delete frontmatter.tags;
    }
  });
});
